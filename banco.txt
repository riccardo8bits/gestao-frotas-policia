CREATE DATABASE banco_frota;
USE banco_frota;

CREATE TABLE IF NOT EXISTS policial(
	id_policial INT AUTO_INCREMENT,
    nome VARCHAR(255) NOT NULL,
	email VARCHAR(255) NOT NULL,
    senha TEXT NOT NULL,
    cargo_patente VARCHAR(25) NOT NULL,
    matricula VARCHAR(50) NOT NULL,
	PRIMARY KEY (id_policial),
    UNIQUE KEY (email)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

CREATE TABLE IF NOT EXISTS viatura(
	id_viatura INT AUTO_INCREMENT,
    modelo VARCHAR(255) NOT NULL,
    placa VARCHAR(255) NOT NULL,
    prefixo VARCHAR(50) NOT NULL,
    ano VARCHAR(10) NOT NULL,
	km_atual FLOAT NOT NULL,
    status_atual ENUM('boa','ruim') NOT NULL,
--    status_turno ENUM('inicio','fim') NOT NULL,
    PRIMARY KEY (id_viatura)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

CREATE TABLE IF NOT EXISTS responsavel(
	id_responsavel INT AUTO_INCREMENT,
    nome VARCHAR(255) NOT NULL,
    cargo_funcao VARCHAR(150) NOT NULL,
	PRIMARY KEY (id_responsavel)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

CREATE TABLE IF NOT EXISTS itens_manutencao(
	id_item INT AUTO_INCREMENT,
    descricao TEXT NOT NULL,
    categoria_falha VARCHAR(255) NOT NULL,
    valor FLOAT NOT NULL,
    PRIMARY KEY (id_item)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

CREATE TABLE IF NOT EXISTS checklist_turno(
	id_checklist INT AUTO_INCREMENT,
    viatura_id INT NOT NULL,
    policial_id INT NOT NULL,
    data_hora DATETIME DEFAULT CURRENT_TIMESTAMP NOT NULL,
    status_pneus ENUM('normal','regular','troca') NOT NULL,
    km_abertura FLOAT NOT NULL,
    relatorio_problemas TEXT NOT NULL,
    PRIMARY KEY (id_checklist),
    CONSTRAINT fk_ckecklist_viatura
		FOREIGN KEY checklist_turno(viatura_id)
		REFERENCES viatura(id_viatura)
        ON DELETE SET NULL,
	CONSTRAINT fk_ckecklist_policial
		FOREIGN KEY checklist_turno(policial_id)
		REFERENCES policial(id_policial)
        ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

CREATE TABLE IF NOT EXISTS ordem_servico(
	id_os INT AUTO_INCREMENT,
    viatura_id INT NOT NULL,
	responsavel_id INT NOT NULL,
    data_abertura DATETIME DEFAULT CURRENT_TIMESTAMP NOT NULL,
	data_fechamento DATETIME,
    tipo_manutencao ENUM('PREVENTIVA', 'CORRETIVA') NOT NULL,
    status_os ENUM('ABERTA','AGUARDANDO','EM_ANDAMENTO','CONCLUIDA','CANCELADA') NOT NULL,
    custo_total FLOAT NOT NULL,
    PRIMARY KEY (id_os),
    CONSTRAINT fk_os_viatura
		FOREIGN KEY ordem_servico(viatura_id)
		REFERENCES viatura(id_viatura)
        ON DELETE SET NULL,
	CONSTRAINT fk_os_responsavel
		FOREIGN KEY ordem_servico(responsavel_id)
		REFERENCES responsavel(id_responsavel)
        ON DELETE SET NULL
    
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

CREATE TABLE IF NOT EXISTS ordem_iten(
	id_ordemItem INT AUTO_INCREMENT,
    os_id INT NOT NULL,
    item_id INT NOT NULL,
    CONSTRAINT fk_ordemItem_os
		FOREIGN KEY (os_id)
		REFERENCES ordem_servico(id_os)
        ON DELETE SET NULL,
	CONSTRAINT fk_ordemItem_itens
		FOREIGN KEY (item_id)
		REFERENCES itens_manutencao(id_item)
        ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
